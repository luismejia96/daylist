    Database setup (one-time)

    In your Supabase project, open SQL Editor and run this once. Row-level security keeps each account’s list private.

    create table if not exists public.daylist_lists (
  user_id uuid primary key references auth.users(id) on delete cascade,
  tasks jsonb not null default '[]'::jsonb,
  updated_at timestamptz not null default now()
);

alter table public.daylist_lists enable row level security;
grant select, insert, update, delete on public.daylist_lists to authenticated;

drop policy if exists "Users manage own Daylist" on public.daylist_lists;
create policy "Users manage own Daylist"
on public.daylist_lists for all to authenticated
using ((select auth.uid()) = user_id)
with check ((select auth.uid()) = user_id);
